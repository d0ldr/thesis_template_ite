\documentclass[english,tgmek,plan]{hhthesis}

%% Above, choose:
%%   - the report you are writing:  plan, midterm or final
%%       plan    = project plan      (Sv: projektplan)
%%       midterm = half-time report  (Sv: halvtidsrapport)
%%       final   = final thesis      (Sv: slutrapport)
%%   - your language:  english or swedish
%%   - your programme: tgelt, tgdat, tgmek, tacda, tacis or tacis.elde
%%   - optional: nohints   (hide the template instructions and green Swedish heading names)
%%               justified (justified text instead of left aligned as in the HH Word template)
%% Keep writing in the same files during the whole course: when you go from
%% the plan to the half-time report (and then to the thesis), just change the
%% option and the template adds/removes the chapters and sections needed.

\usepackage{tikz}
\usetikzlibrary{intersections, through}
\usetikzlibrary{datavisualization.formats.functions} % BPA tillägg
\usetikzlibrary{decorations.pathreplacing,calligraphy}
\usetikzlibrary{math}
\usetikzlibrary{calc}
\usepackage{pgfplots} % it is a must when it needs reading csv files
\pgfplotsset{compat=1.18,
            every axis/.append style={
                line width=1pt
            }}

\addbibresource{thesis.bib}

\input{glossary.tex} % add glossary and acronym lists before document

\date{\today}
\authorone{Name1 Surname1}
\authortwo{Name2 Surname2}

\title{The Title of Your Thesis}
\subtitle{Subtitle (optional)}
% The report type ("Project Plan", "Half-time Report") is added to the
% title page automatically. Do not put it in the subtitle.

\begin{document}

% Abstract, Sammanfattning and Preface: final thesis only
\inputin{final}{frontmatter/abstract}
\inputin{final}{frontmatter/sammanfattning}
\inputin{final}{frontmatter/preface}

\tableofcontents

% Lists of figures, tables, listings and acronyms: half-time report and thesis
\begin{notplan}
\listoffigures
\listoftables
\lstlistoflistings

\printglossary[type=\acronymtype] % Print acronyms
\end{notplan}

%                                         plan   midterm   final
\input{chapters/introduction}           %   x       x        x
\inputin{midterm,final}{chapters/technical_background} %     x        x
\input{chapters/method}                 %   x       x        x
\inputin{final}{chapters/implementation} %                   x
\inputin{midterm,final}{chapters/results} %         x        x  (preliminary in midterm)
\inputin{final}{chapters/discussion}    %                    x
\inputin{final}{chapters/conclusion}    %                    x
\inputin{plan,midterm}{chapters/executionplan} % x  x  (with plan follow-up in midterm)

\chapter*{\bibname}
\printbibliography[heading=none]

\IfMode{midterm,final}{%
\appendix
\input{appendices/a-appendix.tex}
%\input{appendices/b-appendix.tex}
}{}

\end{document}
